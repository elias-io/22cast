cmake_minimum_required(VERSION 3.22)
project(22Cast LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(V ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
file(GLOB PLIST_SRC ${V}/libplist/src/*.c ${V}/libplist/libcnary/*.c)
add_library(plist STATIC ${PLIST_SRC})
target_include_directories(plist PUBLIC ${V}/libplist/include PRIVATE ${V}/libplist/src ${V}/libplist/libcnary/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(plist PRIVATE HAVE_CONFIG_H)
file(GLOB CORE_SRC ${V}/uxplay/lib/*.c ${V}/uxplay/lib/playfair/*.c ${V}/uxplay/lib/llhttp/*.c ${V}/uxplay/lib/mdnsd/*.c)
add_library(airplay STATIC ${CORE_SRC})
target_include_directories(airplay PUBLIC ${V}/uxplay/lib PRIVATE ${V}/uxplay/lib/playfair ${V}/uxplay/lib/llhttp ${V}/uxplay/lib/mdnsd)
target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 NOHOLD OPENSSL_API_COMPAT=0x10101000L _GNU_SOURCE)
target_compile_options(airplay PRIVATE -Wno-deprecated-declarations -Wno-pointer-sign)
if(ANDROID)
 set(SSL ${CMAKE_CURRENT_SOURCE_DIR}/prebuilt/${ANDROID_ABI})
 if(NOT EXISTS ${SSL}/lib/libcrypto.a)
  message(FATAL_ERROR "Run python3 scripts/build-native-deps.py first")
 endif()
 add_library(crypto STATIC IMPORTED)
 set_target_properties(crypto PROPERTIES IMPORTED_LOCATION ${SSL}/lib/libcrypto.a INTERFACE_INCLUDE_DIRECTORIES ${SSL}/include)
 target_compile_definitions(airplay PRIVATE STANDALONE TARGET_POSIX _LINUX)
 target_link_libraries(airplay PUBLIC plist crypto dl m)
 add_library(cast22 SHARED receiver.cpp android_jni.cpp)
 add_library(avcodec STATIC IMPORTED)
 add_library(avutil STATIC IMPORTED)
 set(MEDIA ${SSL}/ffmpeg)
 set_target_properties(avcodec PROPERTIES IMPORTED_LOCATION ${MEDIA}/lib/libavcodec.a INTERFACE_INCLUDE_DIRECTORIES ${MEDIA}/include)
 set_target_properties(avutil PROPERTIES IMPORTED_LOCATION ${MEDIA}/lib/libavutil.a INTERFACE_INCLUDE_DIRECTORIES ${MEDIA}/include)
 target_link_options(cast22 PRIVATE "-Wl,-z,max-page-size=16384")
 target_link_libraries(cast22 airplay avcodec avutil log android)
else()
 find_package(OpenSSL 3 REQUIRED)
 find_package(Threads REQUIRED)
 target_link_libraries(airplay PUBLIC plist OpenSSL::Crypto Threads::Threads)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale sdl2)
 add_executable(cast22 receiver.cpp desktop.cpp)
 target_link_libraries(cast22 PRIVATE airplay PkgConfig::AV)
endif()
